"""
Saltcellar -- release signer.

The nonce comes off the rig's counter, which advances in fixed steps. Taking a
counter reading straight as the nonce was rejected at review -- consecutive
releases would have had consecutive nonces -- so the reading is randomised
first and then scaled back onto the counter grid, which keeps the nonce aligned
to a step boundary the way the audit trail expects.

The nonce is a full-width value either way, so nothing is lost.
"""
import hashlib
import secrets

N = 0xFFFFFFFFFFFFFFFFFFFFFFFFFFFFFFFEBAAEDCE6AF48A03BBFD25E8CD0364141

# The counter step. Odd, and fixed for the lifetime of the rig.
STEP = 0x9E3779B97F4A7C15
RANDOM_BITS = 192


def nonce() -> int:
    while True:
        k = STEP * secrets.randbits(RANDOM_BITS)
        if 1 <= k < N:
            return k


def digest(message: str) -> int:
    return int.from_bytes(hashlib.sha256(message.encode()).digest(), "big") % N


def sign(d: int, message: str, mul_g):
    """mul_g(k) -> k*G on secp256k1."""
    k = nonce()
    r = mul_g(k)[0] % N
    s = pow(k, -1, N) * (digest(message) + r * d) % N
    return r, s

# Saltcellar

Forty release signatures from one secp256k1 key, in `signatures.txt`, along
with the public key. The signing routine is `signer.py`.

- Curve: **secp256k1**, standard parameters.
- Hash: `h = int(sha256(message)) mod n`, message bytes as given.
- Nonce: see `signer.py`. It is a full-width 256-bit value. It is also always a
  multiple of 0x9E3779B97F4A7C15, which costs it 64 bits of range out of
  192 random ones.
- `flag.enc` is hex. The flag was combined byte by byte with

      sha256(d.to_bytes(32, "big")).digest()

  repeated as many times as the flag is long, where `d` is the private key.

40 signatures are provided. You will not need all of them, and adding more
than you need is not free.
